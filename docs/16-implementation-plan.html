CREATE TABLE dream_sessions (
  id              VARCHAR(191) PRIMARY KEY,
  public_id       VARCHAR(191) NOT NULL UNIQUE,
  creator_id      VARCHAR(191) NOT NULL,
  prompt          TEXT NOT NULL,
  title           VARCHAR(160) NULL,
  visibility      ENUM('PUBLIC','CIRCLE','PRIVATE') DEFAULT 'PUBLIC',
  scene_spec      JSON NOT NULL,
  status          ENUM('OPEN','GENERATING','READY','FAILED') DEFAULT 'OPEN',
  created_at      DATETIME(3) DEFAULT CURRENT_TIMESTAMP(3),
  updated_at      DATETIME(3) NOT NULL,
  INDEX idx_dream_creator (creator_id),
  INDEX idx_dream_created (created_at)
);

CREATE TABLE dream_assets (
  id              BIGINT AUTO_INCREMENT PRIMARY KEY,
  dream_id        VARCHAR(191) NOT NULL,
  kind            ENUM('ENV','VIDEO','MUSIC','PROP') NOT NULL,
  provider        VARCHAR(40) NOT NULL,
  provider_job_id VARCHAR(191) NULL,
  status          ENUM('QUEUED','RUNNING','READY','FAILED') DEFAULT 'QUEUED',
  url             VARCHAR(2048) NULL,
  metadata        JSON NULL,
  created_at      DATETIME(3) DEFAULT CURRENT_TIMESTAMP(3),
  INDEX idx_dream_assets_dream (dream_id)
);

CREATE TABLE mood_snapshots (
  id              BIGINT AUTO_INCREMENT PRIMARY KEY,
  scope           ENUM('GLOBAL','REGION','TOPIC') NOT NULL,
  scope_key       VARCHAR(120) NOT NULL,
  mood            ENUM('JOY','CALM','SAD','ANGRY','ANXIOUS') NOT NULL,
  score           DECIMAL(5,4) NOT NULL,
  sample_count    INT NOT NULL,
  window_started  DATETIME(3) NOT NULL,
  window_ended    DATETIME(3) NOT NULL,
  INDEX idx_mood_scope_window (scope, scope_key, window_ended)
);
src/modules/dream/service/dream.service.ts
async createDream(userId: string, input: CreateDreamDto) {
  const sceneSpec = dreamPromptCompiler.compile({
    prompt: input.prompt,
    seed: randomSeed(),
    participants: input.inviteUserIds.length + 1,
  });

  const dream = await prisma.dreamSession.create({
    data: {
      creatorId: userId,
      prompt: input.prompt,
      visibility: input.visibility,
      sceneSpec,
      status: 'OPEN',
      participants: { create: { userId, role: 'HOST' } },
    },
  });

  // Heavy generation is async. Client can enter the room now.
  await dreamGenerationQueue.add('generate-assets', {
    dreamId: dream.id,
    prompt: input.prompt,
    sceneSpec,
    assets: ['ENV', 'VIDEO', 'MUSIC'],
  }, { jobId: `dream:${dream.id}:v1` });

  await pusherService.trigger(`dream-room:${dream.publicId}`, 'room-ready', {
    dreamId: dream.publicId,
    sceneSpec,
  });

  return { dreamId: dream.publicId, sceneSpec };
}

Risk Table

Plan này cố tình tách hiệu ứng wow khỏi core path để không biến API tạo dream thành một request chờ AI nhiều giây.

Risk
Severity
Mitigation
Generative video/music latency cao, provider timeout hoặc quota cạn.
HIGH
Luôn trả sceneSpec trước. Asset jobs retry/backoff, fallback procedural scene + ambient loop. Provider adapter có circuit breaker.
Pusher fan-out quá mạnh khi room đông hoặc pulse đạt ngưỡng lớn.
HIGH
Bucket Redis theo 1s/10s, debounce broadcast, chỉ gửi event aggregate. Cooldown per scope và per user.
Sentiment sai ngữ cảnh, biến app thành mưa/hoa phản cảm trong thời điểm nhạy cảm.
MED
Dùng confidence threshold, moderation override, kill switch emotion_v1, và allowlist mood transition cho sự kiện nhạy cảm.
Nội dung dream prompt tạo môi trường không an toàn.
HIGH
Moderate prompt trước khi enqueue, policy tags trong dream_safety_checks, block hoặc downgrade style trước provider call.
Rung/nhạc tự phát gây khó chịu hoặc vi phạm policy thiết bị/browser.
MED
Opt-in rõ ràng, user setting sensoryEffectsEnabled, silent mode mặc định cho web, mobile deep link tự quyết haptic.
Chi phí AI tăng theo số dream public.
MED
Daily quota, cache theo prompt hash/seed, generate asset chất lượng thấp trước, nâng cấp asset theo engagement.
